cmake_minimum_required(VERSION 3.12)
if(NOT DEFINED ENV{VITASDK})
 message(FATAL_ERROR "VITASDK is required")
endif()
include("$ENV{VITASDK}/share/vita.toolchain.cmake" REQUIRED)
project(im2_exec_probe C)
include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)
set(VITA_APP_NAME "IM2 VM Cache Diagnostic")
set(VITA_TITLEID "IM2V00001")
set(VITA_VERSION "01.00")
add_executable(im2_exec_probe main.c)
target_compile_options(im2_exec_probe PRIVATE -Wall -Wextra -Werror -marm)
target_link_libraries(im2_exec_probe SceLibKernel_stub SceLibc_stub SceIofilemgr_stub)
option(IM2_PACKAGE_VM_PROBE "Explicit standalone VM diagnostic packaging" OFF)
if(IM2_PACKAGE_VM_PROBE)
 vita_create_self(eboot.bin im2_exec_probe)
 vita_create_vpk(im2_exec_probe.vpk ${VITA_TITLEID} eboot.bin VERSION ${VITA_VERSION} NAME ${VITA_APP_NAME})
endif()
