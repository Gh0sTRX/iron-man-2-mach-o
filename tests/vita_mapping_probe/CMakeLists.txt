cmake_minimum_required(VERSION 3.12)
if(NOT DEFINED ENV{VITASDK})
  message(FATAL_ERROR "Set VITASDK to your VitaSDK installation")
endif()
include("$ENV{VITASDK}/share/vita.toolchain.cmake" REQUIRED)
project(im2_mapping_probe C)
include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)
set(VITA_APP_NAME "IM2 Mapping Probe")
set(VITA_TITLEID "IM2M00001")
set(VITA_VERSION "01.00")
add_executable(im2_mapping_probe src/main.c)
target_link_libraries(im2_mapping_probe SceLibKernel_stub SceLibc_stub)
vita_create_self(eboot.bin im2_mapping_probe)
vita_create_vpk(im2_mapping_probe.vpk ${VITA_TITLEID} eboot.bin VERSION ${VITA_VERSION} NAME ${VITA_APP_NAME})
